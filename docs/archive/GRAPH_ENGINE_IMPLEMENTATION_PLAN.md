```dart
class RunInstructionNode extends CompositeNode {
  @override
  Future<dynamic> execute(RunContext context, ToolRegistry tools, IGraphRepository graphRepo) async {
    // 1. Загрузить InstructionGraph
    final instructionGraph = await graphRepo.loadGraph(instructionBlueprintId);

    // 2. Создать новый контекст (клон)
    final subContext = context.clone();

    // 3. Скопировать переменные согласно inputMapping
    for (final entry in inputMapping.entries) {
      final value = context.getVar(entry.value);
      subContext.setVar(entry.key, value);
    }

    // 4. Выполнить InstructionRunner
    final runner = InstructionRunner(
      graph: instructionGraph,
      context: subContext,
      tools: tools,
      graphRepo: graphRepo,
      runRepo: runRepo,
    );
    await runner.run();

    // 5. Скопировать результаты согласно outputMapping
    for (final entry in outputMapping.entries) {
      final value = subContext.getVar(entry.key);
      context.setVar(entry.value, value);
    }
  }
}
```

#### 3.1 Параллельное выполнение (Branches)

**Файл:** `pkgs/aq_graph_engine/lib/src/runners/polymorphic_workflow_runner.dart`

**Изменения:**
```dart
class PolymorphicWorkflowRunner {
  final Map<String, Set<String>> _branchVisitedEdges = {};  // branch -> visited edges
  final Map<String, RunContext> _branchContexts = {};        // branch -> context

  Future<void> _processNode(IWorkflowNode node, int depth, RunContext context) async {
    // Получить исходящие рёбра
    final outgoingEdges = graph.edges.values
        .where((e) => e.sourceId == node.id)
        .toList();

    if (outgoingEdges.isEmpty) return;

    // Группировать по branchName
    final branchesByName = <String, List<WorkflowEdge>>{};
    for (final edge in outgoingEdges) {
      branchesByName.putIfAbsent(edge.branchName, () => []).add(edge);
    }

    // Если несколько веток - запустить параллельно
    if (branchesByName.length > 1) {
      await Future.wait(
        branchesByName.entries.map((entry) async {
          final branchName = entry.key;
          final edges = entry.value;

          // Клонировать контекст для ветки
          final branchContext = context.clone();
          branchContext.currentBranch = branchName;
          _branchContexts[branchName] = branchContext;

          // Выполнить ветку
          for (final edge in edges) {
            final nextNode = graph.nodes[edge.targetId];
            if (nextNode != null) {
              await _processNode(
                WorkflowNodeFactory.fromJson(nextNode.toJson()),
                depth + 1,
                branchContext,
              );
            }
          }
        }),
      );
    } else {
      // Одна ветка - выполнить последовательно
      for (final edge in outgoingEdges) {
        final nextNode = graph.nodes[edge.targetId];
        if (nextNode != null) {
          await _processNode(
            WorkflowNodeFactory.fromJson(nextNode.toJson()),
            depth + 1,
            context,
          );
        }
      }
    }
  }
}
```

#### 3.2 Синхронизация веток (Merge Nodes)

**Логика:**
```dart
Future<bool> _canExecuteNode(IWorkflowNode node) async {
  // Получить все входящие рёбра
  final incomingEdges = graph.edges.values
      .where((e) => e.targetId == node.id)
      .toList();

  if (incomingEdges.length <= 1) {
    return true;  // Не merge node
  }

  // Проверить что все входящие рёбра посещены
  for (final edge in incomingEdges) {
    final branchName = edge.branchName;
    if (!_branchVisitedEdges[branchName]?.contains(edge.id) ?? true) {
      return false;  // Ещё не все ветки дошли
    }
  }

  return true;  // Все ветки готовы
}
```

#### 3.3 Условные переходы (Conditional Edges)

**Файл:** `pkgs/aq_graph_engine/lib/src/engine/condition_evaluator.dart`

**Функциональность:**
```dart
class ConditionEvaluator {
  /// Вычислить условие
  /// Поддерживаемые операторы: ==, !=, >, <, >=, <=, contains, startsWith, endsWith
  bool evaluate(String expression, RunContext context) {
    // Парсинг: "user_approved == true"
    final parts = expression.split(' ');
    if (parts.length != 3) return false;

    final varName = parts[0];
    final operator = parts[1];
    final targetValue = parts[2];

    final actualValue = context.getVar(varName);

    return _compare(actualValue, operator, targetValue);
  }

  bool _compare(dynamic actual, String operator, String target) {
    // Реализация операторов
    switch (operator) {
      case '==': return actual.toString() == target;
      case '!=': return actual.toString() != target;
      case '>': return (actual as num) > num.parse(target);
      // ... остальные операторы
      default: return false;
    }
  }
}
```

**Использование в Runner:**
```dart
Future<void> _processNode(IWorkflowNode node, int depth, RunContext context) async {
  await node.execute(context, _tools, _graphRepo);

  final outgoingEdges = graph.edges.values
      .where((e) => e.sourceId == node.id)
      .toList();

  for (final edge in outgoingEdges) {
    // Проверить условие
    if (edge.type == WorkflowEdgeType.conditional) {
      final condition = edge.conditionExpression ?? '';
      if (!_conditionEvaluator.evaluate(condition, context)) {
        continue;  // Пропустить это ребро
      }
    }

    // Выполнить следующий узел
    final nextNode = graph.nodes[edge.targetId];
    if (nextNode != null) {
      await _processNode(
        WorkflowNodeFactory.fromJson(nextNode.toJson()),
        depth + 1,
        context,
      );
    }
  }
}
```

#### 3.4 Обработка ошибок (onError Edges)

**Изменения в Runner:**
```dart
Future<void> _processNode(IWorkflowNode node, int depth, RunContext context) async {
  try {
    await node.execute(context, _tools, _graphRepo);

    // Успех - следовать onSuccess рёбрам
    final successEdges = graph.edges.values
        .where((e) => e.sourceId == node.id && e.type == WorkflowEdgeType.onSuccess)
        .toList();

    for (final edge in successEdges) {
      final nextNode = graph.nodes[edge.targetId];
      if (nextNode != null) {
        await _processNode(
          WorkflowNodeFactory.fromJson(nextNode.toJson()),
          depth + 1,
          context,
        );
      }
    }
  } catch (e) {
    // Ошибка - следовать onError рёбрам
    _log('❌ Node Error: $e');
    context.setVar('last_error', e.toString());

    final errorEdges = graph.edges.values
        .where((e) => e.sourceId == node.id && e.type == WorkflowEdgeType.onError)
        .toList();

    if (errorEdges.isEmpty) {
      rethrow;  // Нет обработчика - пробросить ошибку
    }

    for (final edge in errorEdges) {
      final nextNode = graph.nodes[edge.targetId];
      if (nextNode != null) {
        await _processNode(
          WorkflowNodeFactory.fromJson(nextNode.toJson()),
          depth + 1,
          context,
        );
      }
    }
  }
}
```

```dart
class SuspendResumeManager {
  final IRunRepository _runRepo;

  /// Приостановить выполнение
  Future<void> suspend({
    required String runId,
    required String nodeId,
    required RunContext context,
    required List<String> logs,
    required Map<String, dynamic> engineState,
  }) async {
    await _runRepo.suspendRun(
      runId: runId,
      contextJson: jsonEncode({
        'user_context': context.toJson(),
        'engine_state': engineState,
      }),
      nodeId: nodeId,
      logs: logs,
    );
  }

  /// Возобновить выполнение
  Future<RunContext> resume({
    required String runId,
    required Map<String, dynamic> injectedVariables,
  }) async {
    final savedRun = await _runRepo.getRun(runId);
    final contextJson = savedRun['context_json'] as String;
    final parsedState = jsonDecode(contextJson);

    // Восстановить контекст
    final context = RunContext.fromJson(
      parsedState['user_context'],
      (msg, {type, depth, branch, details}) => print(msg),
    );

    // Добавить новые данные
    context.state.addAll(injectedVariables);

    return context;
  }
}
```

#### 3.8 Retry Logic

**Файл:** `pkgs/aq_graph_engine/lib/src/nodes/base/automatic_node.dart`

**Добавить:**
```dart
abstract class AutomaticNode extends IWorkflowNode {
  /// Количество попыток при ошибке
  int get maxRetries => 0;

  /// Задержка между попытками
  Duration get retryDelay => Duration(seconds: 1);

  @override
  Future<dynamic> execute(RunContext context, ToolRegistry tools, IGraphRepository graphRepo) async {
    int attempt = 0;

    while (attempt <= maxRetries) {
      try {
        return await executeImpl(context, tools, graphRepo);
      } catch (e) {
        attempt++;
        if (attempt > maxRetries) {
          rethrow;
        }

        context.log('⚠️ Retry $attempt/$maxRetries after error: $e');
        await Future.delayed(retryDelay);
      }
    }
  }

  /// Реализация выполнения (переопределяется в подклассах)
  Future<dynamic> executeImpl(RunContext context, ToolRegistry tools, IGraphRepository graphRepo);
}
```

1. **Сложность параллельного выполнения** - Mitigation: Тщательное тестирование, использование Future.wait
2. **Race conditions в merge nodes** - Mitigation: Атомарные операции, locks
3. **Утечки памяти в long-running workflows** - Mitigation: Profiling, cleanup после каждого узла
4. **Проблемы с сериализацией контекста** - Mitigation: Тесты на suspend/resume
